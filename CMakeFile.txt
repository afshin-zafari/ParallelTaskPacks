cmake_minimum_required(VERSION 3.15)
project(InferenceTests VERSION 0.1 LANGUAGES CXX)

# C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Default to Release if not specified
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# Collect sources (root-level .cpp files and subfolders)
file(GLOB_RECURSE PROJECT_SOURCES
  "${PROJECT_SOURCE_DIR}/*.cpp"
  "${PROJECT_SOURCE_DIR}/*.cxx"
  "${PROJECT_SOURCE_DIR}/*.cc"
)

# Create executable
add_executable(${PROJECT_NAME} ${PROJECT_SOURCES})

# Ensure project root is available for includes
target_include_directories(${PROJECT_NAME} PRIVATE "${PROJECT_SOURCE_DIR}")

# MSVC: enable warnings and sensible defaults
if(MSVC)
  target_compile_options(${PROJECT_NAME} PRIVATE /W4 /permissive-)
else()
  target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
  # On non-MSVC platforms, enable position independent code for potential shared libs.
  set_target_properties(${PROJECT_NAME} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endif()

# Convenience: print number of source files detected
message(STATUS "InferenceTests: ${PROJECT_NAME} sources: ${PROJECT_SOURCES}")
